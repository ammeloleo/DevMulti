# Script SQL

```sql
-- Criação da tabela de Usuários
CREATE TABLE usuario (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome VARCHAR(255) NOT NULL,
    termo VARCHAR(50),
    events JSONB
);

-- Criação da tabela de Postagens (Trabalhos/Artigos)
CREATE TABLE postagens (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    titulo TEXT NOT NULL,
    text TEXT NOT NULL,
    data TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    curtidas INTEGER DEFAULT 0,
    id_usuario BIGINT REFERENCES usuario(id) ON DELETE SET NULL
);

-- Criação da tabela de Eventos (Calendário Acadêmico)
CREATE TABLE eventos (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    date DATE NOT NULL,
    type VARCHAR(50) NOT NULL CHECK (type IN ('prova', 'entrega', 'docente', 'outro')),
    id_usuario BIGINT REFERENCES usuario(id) ON DELETE SET NULL
);
```
